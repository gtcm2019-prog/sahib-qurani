FUNCTION | private.hifz_map
CREATE OR REPLACE FUNCTION private.hifz_map(p_user uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare
  rw text; hp smallint[]; pages int[]; thumns int[]; hizbs int[] := '{}'; h int;
begin
  select riwaya into rw from public.profiles where id = p_user;
  hp := private.hizb_pages(rw);
  select coalesce(array_agg(distinct n order by n), '{}') into pages
    from public.revision_plans p, generate_series(p.start_no::int, p.end_no::int) n
   where p.user_id = p_user and p.kind = 'hifz' and p.unit = 'page' and p.status = 'done';
  select coalesce(array_agg(distinct n order by n), '{}') into thumns
    from public.revision_plans p, generate_series(p.start_no::int, p.end_no::int) n
   where p.user_id = p_user and p.kind = 'hifz' and p.unit = 'thumn' and p.status = 'done';
  for h in 1..60 loop
    if (select count(*) from unnest(thumns) t where t between (h-1)*8+1 and h*8) = 8
       or (select count(*) from unnest(pages) x where x between hp[1][h] and hp[2][h]) = hp[2][h] - hp[1][h] + 1
    then hizbs := hizbs || h; end if;
  end loop;
  return jsonb_build_object('riwaya', coalesce(rw, 'hafs'), 'pages', to_jsonb(pages),
                            'thumns', to_jsonb(thumns), 'hizbs', to_jsonb(hizbs));
end $function$

FUNCTION | private.refresh_assiduity
CREATE OR REPLACE FUNCTION private.refresh_assiduity(p_user uuid)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare
  today date := private.member_today(p_user);
  n_all int; n_done int; rate int; lvl text; streak int := 0; d date;
  day_ok constant text := 'select exists (select 1 from public.revision_plans where user_id = $1 and plan_date = $2)
     and not exists (select 1 from public.revision_plans where user_id = $1 and plan_date = $2 and status <> ''done'')';
  ok boolean;
begin
  select count(*), count(*) filter (where status = 'done') into n_all, n_done
    from public.revision_plans
   where user_id = p_user and plan_date > today - 30
     and (plan_date < today or (plan_date = today and status = 'done'));
  rate := case when n_all > 0 then round(100.0 * n_done / n_all) end;
  lvl  := case when n_all < 7 then 'new' when rate >= 80 then 'top' when rate >= 50 then 'good' else 'low' end;
  execute day_ok into ok using p_user, today;
  d := case when ok then today else today - 1 end;
  loop
    execute day_ok into ok using p_user, d;
    exit when not ok or streak >= 3650;
    streak := streak + 1; d := d - 1;
  end loop;
  update public.profiles set assiduity = lvl, assiduity_rate = rate, revision_streak = streak
   where id = p_user
     and (assiduity, assiduity_rate, revision_streak) is distinct from (lvl, rate, streak);
end $function$

FUNCTION | public.admin_app_stats
CREATE OR REPLACE FUNCTION public.admin_app_stats()
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
begin
  if not private.is_admin() then raise exception 'forbidden'; end if;
  return jsonb_build_object(
    'installs',  (select count(*) from public.app_devices),
    'week',      (select count(*) from public.app_devices where installed_at > now() - interval '7 days'),
    'active30',  (select count(*) from public.app_devices where last_seen > now() - interval '30 days'),
    'inactive30',(select count(*) from public.app_devices where last_seen <= now() - interval '30 days'),
    'members',   (select count(distinct user_id) from public.app_devices where user_id is not null),
    'weekly', coalesce((select jsonb_agg(jsonb_build_array(w, n) order by w) from (
        select to_char(date_trunc('week', installed_at), 'YYYY-MM-DD') w, count(*) n
        from public.app_devices where installed_at > now() - interval '12 weeks' group by 1) x), '[]'::jsonb),
    'monthly', coalesce((select jsonb_agg(jsonb_build_array(m, n) order by m) from (
        select to_char(date_trunc('month', installed_at), 'YYYY-MM') m, count(*) n

FUNCTION | public.app_ping
CREATE OR REPLACE FUNCTION public.app_ping(p_device uuid, p_platform text, p_install boolean DEFAULT false)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
begin
  if p_device is null then return; end if;

    installed_at = case when p_install then now() else coalesce(d.installed_at, now()) end
  where d.last_seen < now() - interval '1 hour' or p_install or (d.user_id is null and auth.uid() is not null);
end $function$

FUNCTION | public.pseudo_status
CREATE OR REPLACE FUNCTION public.pseudo_status(p_name text)
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare
  me    uuid := auth.uid();
  n     text := left(regexp_replace(btrim(coalesce(p_name, '')), '\s+', ' ', 'g'), 50);
  base  text;
  town  text;
  cand  text;
  res   text[] := '{}';
  tries int := 0;
begin
  if me is null then raise exception 'not authenticated'; end if;
  if n = '' then return jsonb_build_object('free', false, 'suggestions', '[]'::jsonb); end if;

  if not exists (select 1 from public.profiles p
                 where lower(btrim(p.first_name)) = lower(n) and p.id <> me) then
    return jsonb_build_object('free', true);
  end if;

  base := left(replace(n, ' ', '_'), 40);
  select left(split_part(btrim(p.city), ' ', 1), 4) into town from public.profiles p where p.id = me;

  foreach cand in array array[
    base || extract(year from now())::int::text,
    case when coalesce(town, '') <> '' then base || '_' || initcap(town) end
  ] loop
    if cand is not null and not exists (select 1 from public.profiles p
        where lower(btrim(p.first_name)) = lower(cand) and p.id <> me) then
      res := res || cand;
    end if;
  end loop;

  while coalesce(array_length(res, 1), 0) < 3 and tries < 40 loop
    tries := tries + 1;
    cand := base || '_' || (10 + floor(random() * 90))::int::text;
    if not (cand = any(res)) and not exists (select 1 from public.profiles p
        where lower(btrim(p.first_name)) = lower(cand) and p.id <> me) then
      res := res || cand;
    end if;
  end loop;

  return jsonb_build_object('free', false, 'suggestions', to_jsonb(res));
end $function$

FUNCTION | public.revision_stats
CREATE OR REPLACE FUNCTION public.revision_stats(p_user uuid DEFAULT NULL::uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare u uuid := coalesce(p_user, auth.uid()); today date; m jsonb; res jsonb := '{}'; k text;
  streak int; d date; ok boolean;
begin
  if auth.uid() is null then raise exception 'auth'; end if;
  if u <> auth.uid() and not private.is_active_partner(u) then raise exception 'forbidden'; end if;
  if not exists (select 1 from public.profiles where id = u) then return null; end if;
  perform private.refresh_assiduity(u);
  today := private.member_today(u);
  m := private.hifz_map(u);
  foreach k in array array['hifz','muraja'] loop
    streak := 0;
    d := case when exists (select 1 from public.revision_plans where user_id = u and kind = k
                            and plan_date = today and status = 'done') then today else today - 1 end;
    while exists (select 1 from public.revision_plans where user_id = u and kind = k and plan_date = d and status = 'done') loop
      streak := streak + 1; d := d - 1; exit when streak >= 3650;
    end loop;
    res := res || jsonb_build_object(k, (select jsonb_build_object(
      'streak', streak,
      'rate30', (select round(100.0 * count(*) filter (where status = 'done') / nullif(count(*), 0))
                   from public.revision_plans where user_id = u and kind = k and plan_date > today - 30
                    and (plan_date < today or status = 'done')),
      'done_total', (select count(*) from public.revision_plans where user_id = u and kind = k and status = 'done'),
      'hizb_total', (select coalesce(sum((end_no - start_no + 1) * case unit when 'hizb' then 1 when 'juz' then 2 else 0 end), 0)
                       from public.revision_plans where user_id = u and kind = k and status = 'done'),
      'days', coalesce((select jsonb_agg(jsonb_build_object('d', plan_date, 's',
                 case when status = 'planned' and plan_date < today then 'missed' else status end) order by plan_date)
               from public.revision_plans where user_id = u and kind = k and plan_date > today - 35), '[]'::jsonb))));
  end loop;
  return (select jsonb_build_object(
    'today', today, 'assiduity', pr.assiduity, 'rate30', pr.assiduity_rate, 'streak', pr.revision_streak,
    'hizb_memorized', pr.hizb_memorized, 'map', m,
    'hifz', res->'hifz', 'muraja', res->'muraja')
    from public.profiles pr where pr.id = u);
end $function$

FUNCTION | public.rls_auto_enable
CREATE OR REPLACE FUNCTION public.rls_auto_enable()
 RETURNS event_trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO 'pg_catalog'
AS $function$
DECLARE
  cmd record;
  tbl name;
BEGIN
  FOR cmd IN
    SELECT *
    FROM pg_event_trigger_ddl_commands()
    WHERE command_tag IN ('CREATE TABLE', 'CREATE TABLE AS', 'SELECT INTO')
      AND object_type IN ('table','partitioned table')
  LOOP
     IF cmd.schema_name IS NOT NULL AND cmd.schema_name IN ('public') AND cmd.schema_name NOT IN ('pg_catalog','information_schema') AND cmd.schema_name NOT LIKE 'pg_toast%' AND cmd.schema_name NOT LIKE 'pg_temp%' THEN
      BEGIN
        SELECT c.relname INTO tbl FROM pg_catalog.pg_class c WHERE c.oid = cmd.objid;
        EXECUTE format('alter table if exists %I.%I enable row level security', cmd.schema_name, tbl);
        RAISE LOG 'rls_auto_enable: enabled RLS on %', cmd.object_identity;
      EXCEPTION
        WHEN OTHERS THEN
          RAISE LOG 'rls_auto_enable: failed to enable RLS on %', cmd.object_identity;
      END;
     ELSE
        RAISE LOG 'rls_auto_enable: skip % (either system schema or not in enforced list: %.)', cmd.object_identity, cmd.schema_name;
     END IF;
  END LOOP;
END;
$function$

FUNCTION | public.save_push_subscription

 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
begin

  values ((select auth.uid()), p_endpoint, p_p256dh, p_auth);
end $function$

FUNCTION | public.set_revision_plan
CREATE OR REPLACE FUNCTION public.set_revision_plan(p_kind text, p_date date, p_unit text, p_start integer, p_end integer)
 RETURNS revision_plans
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare me uuid := auth.uid(); today date; r public.revision_plans;
begin
  if me is null then raise exception 'auth'; end if;
  today := private.member_today(me);
  if p_date is null or p_date not in (today, today + 1) then raise exception 'bad_date' using errcode = 'P0001'; end if;
  if exists (select 1 from public.revision_plans
              where user_id = me and plan_date = p_date and kind = p_kind and status <> 'planned') then
    raise exception 'already_done' using errcode = 'P0001';
  end if;
  insert into public.revision_plans (user_id, plan_date, kind, unit, start_no, end_no)
  values (me, p_date, p_kind, p_unit, p_start, p_end)
  on conflict (user_id, plan_date, kind) do update
     set unit = excluded.unit, start_no = excluded.start_no, end_no = excluded.end_no, updated_at = now()
  returning * into r;
  return r;
end $function$

FUNCTION | public.track_ref_visit
CREATE OR REPLACE FUNCTION public.track_ref_visit(p_code text)
 RETURNS void
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
declare r uuid;
begin
  select user_id into r from public.share_codes where code = upper(coalesce(p_code, ''));
  if r is null or r = auth.uid() then return; end if;
  if (select count(*) from public.ref_visits
      where referrer_id = r and created_at > now() - interval '1 hour') >= 200 then return; end if;
  insert into public.ref_visits (referrer_id) values (r);
end $function$
